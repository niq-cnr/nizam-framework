# Candidate chosen: (a) verbatim quote-fidelity of the seven checklist-layer
# epigraphs against Section 2's layer table (SOURCE rows), character-for-character,
# not merely "citation present somewhere" (which verification item 7 already
# mechanizes structurally). This probes something the pipeline has NOT probed:
# Mode B's own first-occurrence-weakness probe for this file (per the task framing)
# targeted a different concern; contract verification item 7 checks that a
# Section-N citation string appears in each heading-delimited block, but never
# diffs the epigraph's actual QUOTED TEXT against the source table's One-Liner
# column. A generator could satisfy item 7 while silently paraphrasing, truncating,
# or drifting the quoted sentence -- this check closes that gap.
import re

source = open('standard/definition_of_done.md').read()
template = open('templates/DoD.md').read()

table_start = source.index('## 2. The Layered Model')
table_end = source.index('## 3. Step-Done')
table_block = source[table_start:table_end]

rows = {}
for line in table_block.splitlines():
    line = line.strip()
    if not line.startswith('|') or 'Layer' in line or '---' in line:
        continue
    parts = [p.strip() for p in line.strip('|').split('|')]
    if len(parts) != 3:
        continue
    layer, one_liner, authority = parts
    rows[layer] = (one_liner, authority)

epigraph_re = re.compile(r'_"([^"]*)"_\s*\n?\(`standard/definition_of_done\.md`\s+(Section \d+)\)', re.MULTILINE)
matches = []
for m in epigraph_re.finditer(template):
    quote = re.sub(r'\s+', ' ', m.group(1)).strip()
    section = m.group(2)
    matches.append((quote, section))

check_layers = ['Step-Done', 'Handoff-Done', 'Gate-Done', 'Feature-Done', 'Plan-Done', 'Merge-Done', 'Release-Done']
assert len(matches) == 7, f"expected 7 epigraph blocks, found {len(matches)}"

mismatches = []
for (layer, (source_quote, source_section)), (template_quote, template_section) in zip(rows.items(), matches):
    if layer not in check_layers:
        continue
    ok_text = (source_quote == template_quote)
    ok_section = source_section == template_section
    status = 'MATCH' if (ok_text and ok_section) else 'MISMATCH'
    print(f"[{status}] {layer}")
    print(f"  SOURCE   : {source_quote!r}  ({source_section})")
    print(f"  TEMPLATE : {template_quote!r}  ({template_section})")
    if status == 'MISMATCH':
        mismatches.append(layer)

if mismatches:
    print(f"\nFAIL -- {len(mismatches)} mismatching row(s): {mismatches}")
else:
    print("\nPASS -- all 7 checklist-layer epigraphs are byte-for-byte verbatim against "
          "standard/definition_of_done.md Section 2's layer table (quoted text AND cited "
          "Section N), for all 7 rows (task floor was 3; all 7 checked for completeness).")
