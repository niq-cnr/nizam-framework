class Rejection(Exception):
    def __init__(self, code, detail):
        self.code, self.detail = code, detail

def need(ok, code, detail):

def sha(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    p = subprocess.run(['git', '-C', str(ROOT), *args], capture_output=True)
    need(p.returncode == 0, 'setup-git', p.stderr.decode(errors='replace'))
    return p.stdout


class View:
    def __init__(self, overlay=None):
        self.overlay = overlay
        self.cache = {}

    def read(self, path):
        need(not Path(path).is_absolute() and '..' not in Path(path).parts,
             'setup-path', path)
        p = self.overlay / path if self.overlay else None
        if p and p.is_file():
            return p.read_bytes()
        p = ROOT / path
        need(p.is_file(), 'missing-document' if path == DOC else 'setup-missing', path)
        need(not p.is_symlink(), 'scope-symlink', path)
        return p.read_bytes()

    def text(self, path):
        return self.read(path).decode('utf-8')

    def obj(self, path):
        return json.loads(self.read(path))

    def base(self, path, revision=BASE):
        if revision not in BASE_BYTES:
            with tarfile.open(fileobj=io.BytesIO(git('archive', revision))) as archive:
                BASE_BYTES[revision] = {m.name: archive.extractfile(m).read()
                                        for m in archive if m.isfile()}
        need(path in BASE_BYTES[revision], 'setup-base-path', path)
        return BASE_BYTES[revision][path]

    def base_obj(self, path):
        return json.loads(self.base(path))

    def paths(self):
        tracked = set(git('ls-tree', '-r', '--name-only', BASE).decode().splitlines())
        present = {p.relative_to(ROOT).as_posix() for p in ROOT.rglob('*')
                   if p.is_file() and '.git' not in p.relative_to(ROOT).parts}
        if self.overlay:
            present |= {p.relative_to(self.overlay).as_posix() for p in self.overlay.rglob('*')
                        if p.is_file()}
        return tracked, present

def prose(text):
    text = re.sub(r'<!--.*?-->', '', text, flags=re.S)
    text = re.sub(r'(?ms)^```[^\n]*\n.*?^```\s*$', '', text)
    return re.sub(r'\s+', ' ', text)


def markdown(text):
    m = re.match(r'\A---\n(.*?)\n---\n(.*)\Z', text, re.S)
    need(m is not None, 'document-frontmatter', 'missing YAML frontmatter')
    return yaml.safe_load(m[1]), m[2]


def sections(v):

    text = v.text(DOC)
    fm, body = markdown(text)
    need(fm.get('version') == '0.1.0' and fm.get('status') == 'active' and
         fm.get('enforcement') == 'consumer-aspirational' and
         fm.get('authoritative_source') == DOC, 'document-frontmatter', 'binding identity')
    headings = re.findall(r'(?m)^## .*$', body)
    need(headings == c['implementation_requirements']['sections'], 'section-set', 'exact eleven sections/order')
    spans = {}
    for i, heading in enumerate(headings):
        start = body.index(heading) + len(heading)
        end = body.index(headings[i+1]) if i+1 < len(headings) else len(body)
        spans[i+1] = body[start:end]
        need(prose(spans[i+1]).strip(), 'section-empty', heading)
    return spans

def strip_version(v, path):
    old = v.base(path).decode(); new = v.text(path)
    oldfm, _ = markdown(old); fm, _ = markdown(new)
    need(fm.get('version') == VERSIONS[path], 'version-delta', path)
    need(isinstance(fm.get('change_log'), list) and len(fm['change_log']) == len(oldfm['change_log'])+1,
         'changelog-delta', path)
    first = fm['change_log'][0]
    need(first.get('version') == VERSIONS[path] and bool(first.get('date')) and bool(first.get('summary')),
         'changelog-delta', path+' new record')
    need(fm['change_log'][1:] == oldfm['change_log'], 'historical-log', path)
    # Preserve the exact old frontmatter bytes, including logs, after one inserted record.
    entries = list(re.finditer(r'(?m)^  - version:', new.split('\n---\n',1)[0]))
    need(len(entries) >= 2, 'changelog-delta', path+' entries')
    normalized = new[:entries[0].start()] + new[entries[1].start():]
    normalized = re.sub(r'(?m)^version: '+re.escape(VERSIONS[path])+r'$',
                        'version: '+str(oldfm['version']), normalized, count=1)
    need(normalized.split('\n---\n',1)[0] == old.split('\n---\n',1)[0],
         'historical-log', path+' frontmatter bytes')
    return old, normalized


def availability_cut(text):
    start = text.index('# Live Runtime Sessions\n\n') + len('# Live Runtime Sessions\n\n')
    end = text.index('\n\nA runtime session', start)
    first = text[start:end]
    text = text[:start] + '<availability>' + text[end:]
    marker = 'preserve fields they do not own.'
    start = text.index(marker, text.index('## 12. Durable records'))+len(marker)
    end = text.index('\n\n| requirement-id', start)
    second = text[start:end]
    return text[:start]+'<availability>'+text[end:], first+' '+second

def budget(v):

    need(old['original_estimate_lines'] == now['original_estimate_lines'] == 7200 and old['ceiling_130pct'] == now['ceiling_130pct'] == 9360,
         'budget-authority', '7200/9360 unchanged')
    need(now['per_feature'][:len(old['per_feature'])] == old['per_feature'] and
         [x['actual'] for x in old['per_feature']] == [1711,3133], 'budget-history', 'preceding4844')
    for k in ('phase_level_items','counting_rule','phase_base_revision','authorization_revision','phase_level_costs','historical_inflight_measurements'):
        need(now[k] == old[k], 'budget-history', k)
    tracked,present = v.paths(); counted=excluded=evidence=0
    for path in sorted(tracked | present):
        if PHASE_BASE not in BASE_BYTES:
            v.base('NIZAM.json', PHASE_BASE)
        before = BASE_BYTES[PHASE_BASE].get(path,b'').decode(errors='replace').splitlines()
        after = v.read(path).decode(errors='replace').splitlines() if path in present else []
        if before == after: continue
        lines = sum((a1-a0)+(b1-b0) for tag,a0,a1,b0,b1 in difflib.SequenceMatcher(None,before,after,autojunk=False).get_opcodes() if tag!='equal')
        if path.startswith('.agent/evidence/') or (path.startswith('.agent/validator/') and not path.endswith('.json')): evidence+=lines
        elif path == '.agent/run_state.json': excluded+=lines
        else:
            need(path in PRODUCTS+PARENT or path == '.agent/contracts/112.json' or re.fullmatch(r'\.agent/(qa|validator)/112(?:-[\w-]+)?\.json',path), 'budget-unclassified',path)
            counted+=lines
    need(4844+counted <= 9360 and counted <= 7266, 'budget-ceiling', str(counted))
    complete = next(x for x in v.obj('.agent/feature_list_015.json')['features'] if x['id']=='112')['status']=='complete'
    need(len(now['per_feature']) == len(old['per_feature'])+(1 if complete else 0), 'budget-history','only112 completion append')
    if complete:
        entry=now['per_feature'][-1]
        need(entry['id']=='112' and entry['estimated']==1200 and entry['actual']==counted and now['total_lines_changed']==4844+counted,
             'budget-measurement','actual completed112 measurement')
    else: need(now['total_lines_changed']==4844,'budget-history','in-flight not appended')
    print(json.dumps({'base':PHASE_BASE,'counted':counted,'evidence':evidence,'excluded':excluded,'prior_completed':4844,'remaining_before112':4516,'prior_only_3x':7266},sort_keys=True))

    need(set(inputs['source_sha256'])==set(PRODUCTS+['.agent/contracts/112.json']), 'candidate-binding','exact ten source inputs')
    need(set(inputs['verification_sha256'])==set(CHECKER_FILES), 'checker-binding','exact independent bundle')
    for group in ('source_sha256','verification_sha256','immutable_input_sha256'):
        for path,digest in inputs[group].items():
            need(sha(v.read(path))==digest, 'source-binding' if group=='source_sha256' else 'checker-binding' if group=='verification_sha256' else 'historical-immutable',path)
    need(inputs['mode_b_sha256']==sha(v.read('.agent/validator/112-mode-b.json')), 'candidate-binding','ModeB hash')
    modeb=v.obj('.agent/validator/112-mode-b.json')
    need(approved_block(modeb) and modeb.get('reviewed_sha256')==inputs['source_sha256'], 'candidate-binding','ModeB exact source map')
    actors=[inputs.get(k) for k in ('generator','test_author','checker_reviewer','acceptance_reviewer')]
    need(all(isinstance(x,str) and x for x in actors) and len(set(actors))==4, 'review-independence','four distinct actors')
    need(inputs.get('primary_sources_sha256')==sha(v.read(E+'primary-sources.json')), 'primary-source-binding','source manifest')
    need(inputs['immutable_input_sha256'].get(E+'approved-base.json')==sha(v.read(E+'approved-base.json')), 'candidate-binding','approved-base receipt')
    need(inputs['immutable_input_sha256'].get(E+'evidence_gate.py')==sha(v.read(E+'evidence_gate.py')), 'candidate-binding','assessed evidence capture adapter')
    tracked,_=v.paths()
    protected=[p for p in tracked if p.startswith(E) or re.fullmatch(r'\.agent/(?:qa|validator)/112(?:-[\w-]+)?\.json',p)]
    need(set(protected)<=set(inputs['immutable_input_sha256'])|set(inputs['source_sha256']), 'historical-immutable','all prior112 gates/raw inputs covered')


def semantic_review(v):
    inputs=seal(v); review=v.obj(E+'semantic-review.json'); sources=v.obj(E+'primary-sources.json')
    need(review.get('reviewer')==inputs['acceptance_reviewer'] and review.get('personally_reviewed') is True,
         'semantic-independence','fresh acceptance actor personally reviewed')
    need(review.get('candidate_sha')==inputs['candidate_sha'] and review.get('source_sha256')==inputs['source_sha256']
         and review.get('qa_inputs_sha256')==sha(v.read(E+'qa-inputs.json')), 'semantic-binding','exact candidate/source/seal')
    need(approved_block(review), 'semantic-verdict','approved with all blocker arrays empty')
    need(sources.get('collector')==review['reviewer'] and isinstance(sources.get('sources'),list), 'primary-source-binding','personally fetched manifest')
    covered=set(); records={}
    for source in sources['sources']:
        host=urlparse(source['url']).hostname
        need(host in ('www.lispworks.com','www.sbcl.org','sbcl.org','asdf.common-lisp.dev') and source.get('http_status')==200
             and bool(source.get('accessed_at')) and source['path'].startswith(E+'primary/')
             and sha(v.read(source['path']))==source['sha256'] and len(v.read(source['path']))>100,
             'primary-source-binding',source.get('url','missing URL'))
        need(source['url'] not in records,'primary-source-binding','duplicate source URL')
        records[source['url']]=source;covered.update(source.get('topics',[]))
    required=set(v.obj(E+'controls.json')['semantic_topics'])
    need(required<=covered,'semantic-coverage','all official-source topics')
    findings=review.get('findings',[])
    need({x.get('topic') for x in findings}==required,'semantic-coverage','exact review topic set')
    for finding in findings:
        need(finding.get('approved') is True and bool(finding.get('section')) and bool(finding.get('analysis'))
             and finding.get('source_urls') and all(u in records for u in finding['source_urls']),
             'semantic-coverage',finding.get('topic','unnamed'))
        need(finding['section'] in contract(v)['implementation_requirements']['sections'], 'semantic-coverage','actual section heading')
    for path,digest in inputs['source_sha256'].items():
        need(sha(v.read(path))==digest,'semantic-binding',path+' changed after review seal')
    print('Semantic receipt checked for completeness and source binding; semantics personally assessed by '+review['reviewer'])


def negative(v):
    # Positive candidate required before mutations: setup errors cannot stand in for a rejection.
    structure(v); scope(v); historical(v); lifecycle(v); budget(v)
    controls=v.obj(E+'controls.json')
    dispatch={'structure':structure,'scope':scope,'historical':historical,'lifecycle':lifecycle,'budget':budget}
    for control in controls['mutations']:
        with tempfile.TemporaryDirectory(prefix='nizam112-control-') as tmp:
            overlay=Path(tmp); target=overlay/control['path']; target.parent.mkdir(parents=True,exist_ok=True)
            original=v.read(control['path']) if control['kind']!='extra' else None
            borrowed={control['path']:original}
            if control['kind']=='section-comment':
                text=original.decode();heading=control['heading'];start=text.index(heading)+len(heading);m=re.search(r'(?m)^## ',text[start:]);end=start+m.start() if m else len(text)
                mutated=(text[:start]+'\n<!--'+text[start:end]+'-->\n'+text[end:]).encode()
            elif control['kind']=='replace':
                text=original.decode();need(text.count(control['old'])>=1,'control-anchor',control['id']);mutated=text.replace(control['old'],control['new'],1).encode()
            elif control['kind'] in ('move-token','insert-section'):
                text=original.decode();heading=control['heading'];start=text.index(heading)+len(heading);m=re.search(r'(?m)^## ',text[start:]);end=start+m.start() if m else len(text)
                span=text[start:end]
                if control['kind']=='move-token':
                    need(control['token'] in span,'control-anchor',control['id'])
                    span=span.replace(control['token'],'REMOVED_TOKEN')
                    suffix='\n<!--misplaced token '+control['token']+'-->\n'
                else:
                    span+='\n'+control['text']+'\n';suffix=''
                mutated=(text[:start]+span+text[end:]+suffix).encode()
            elif control['kind']=='json':
                obj=json.loads(original);where=obj
                for key in control['keys'][:-1]: where=where[key]
                where[control['keys'][-1]]=control['value'];mutated=json.dumps(obj).encode()
            elif control['kind']=='premature-completion':
                for path in PARENT:
                    borrowed[path]=v.read(path)
                    obj=yaml.safe_load(borrowed[path]) if path.endswith('.yaml') else json.loads(borrowed[path])
                    if path.endswith('feature_list_015.json'):
                        f=next(x for x in obj['features'] if x['id']=='112');f.update(status='complete',passes=True)
                    elif path.endswith('phase_015.yaml'):
                        next(x for x in obj['steps'] if x['id']=='112')['status']='COMPLETED'
                    elif path.endswith('backlog_dag.json'):
                        f=next(x for x in obj['packets'] if x['id']=='NIP3-112');f['status']='complete';f['prerequisites']['evidence']['satisfied']=True
                        next(x for x in obj['packets'] if x['id']=='NIP3-113')['prerequisites']['technical']['satisfied']=True
                    elif path.endswith('run_state.json'):
                        obj['current_feature']='113';obj['active_contract_id']=None;obj['completed_contract_ids'].append('112')
                    p=overlay/path;p.parent.mkdir(parents=True,exist_ok=True)
                    p.write_bytes((yaml.safe_dump(obj,sort_keys=False) if path.endswith('.yaml') else json.dumps(obj)).encode())
                mutated=target.read_bytes()
            elif control['kind']=='extra': mutated=b'undeclared112 source\n'
            else: raise Rejection('control-definition',control['id'])
            target.write_bytes(mutated); private=View(overlay)
            try: dispatch[control['mode']](private)
            except Rejection as exc:
                need(exc.code==control['diagnostic'],'control-diagnostic',control['id']+' expected '+control['diagnostic']+' observed '+exc.code)
                print('CONTROL '+control['id']+' EXIT:1 diagnostic:'+exc.code)
            else: raise Rejection('control-false-pass',control['id'])
            for path,data in borrowed.items():
                if data is None: (overlay/path).unlink()
                else: (overlay/path).write_bytes(data)
            dispatch[control['mode']](View(overlay))
            need(original is None or target.read_bytes()==original,'control-restoration',control['id'])
            print('RESTORED '+control['id']+' EXIT:0')


def main():
    parser=argparse.ArgumentParser()
    modes={'structure':structure,'semantic-review':semantic_review,'scope':scope,'historical':historical,
           'lifecycle':lifecycle,'budget':budget,'negative':negative,'integrity':integrity}
    group=parser.add_mutually_exclusive_group(required=True)
    for mode in modes: group.add_argument('--'+mode,action='store_true')
    args=parser.parse_args()
    mode=next(k for k in modes if getattr(args,k.replace('-','_')))
    try:
        modes[mode](View()); print('PASS112 '+mode); return 0
    except Rejection as exc:
        print('FAIL112 '+exc.code+': '+exc.detail); return 1
    except (KeyError,ValueError,TypeError,OSError,yaml.YAMLError) as exc:
        print('FAIL112 setup-error: '+type(exc).__name__+': '+str(exc)); return 2


if __name__=='__main__':
    sys.exit(main())
