def section(body, number, title=None):
    heads = list(re.finditer(r'(?m)^## (\d+)\. ([^\n]+)\n', body))
    selected = [(i, h) for i, h in enumerate(heads) if int(h.group(1)) == number]
    require(len(selected) == 1, 'SECTION:' + str(number))
    i, h = selected[0]
    if title:
        require(h.group(2) == title, 'SECTION_TITLE:' + str(number))
    return body[h.end():heads[i + 1].start() if i + 1 < len(heads) else len(body)]


def escaped(text, position):
    start = position
    while start and text[start - 1] == '\\':
        start -= 1
    return (position - start) % 2 == 1


def blank_markup(text):
    # Keep boundaries/newlines: removing hidden text must not join two fragments
    # into a new synthetic link that did not exist in the original source.
    return ''.join('\n' if char == '\n' else ' ' for char in text)


def visible_link_source(body):
    """Conservative supported source regions, before decoding any entities.

    Simple unindented Markdown paragraphs and standalone HTML anchors qualify.
    Comments, fenced/indented code and exact-delimiter inline code do not.
    Reference links, nested Markdown brackets and Markdown mixed with raw HTML
    in one paragraph are deliberately unsupported, never guessed into a PASS.
    """
    body = re.sub(r'<!--(?s:.*?)(?:-->|\Z)', lambda m: blank_markup(m.group()), body)
    lines, fence_char, fence_size = [], None, 0
    for line in body.splitlines(keepends=True):
        opening = re.match(r'^ {0,3}(`{3,}|~{3,})([^\n]*)', line)
        if fence_char:
            close = re.match(r'^ {0,3}(' + re.escape(fence_char) + r'{'+str(fence_size)+r',})[ \t]*(?:\n|$)', line)
            lines.append(blank_markup(line))
            if close:
                fence_char, fence_size = None, 0
        elif opening:
            fence_char, fence_size = opening.group(1)[0], len(opening.group(1))
            lines.append(blank_markup(line))
        elif line.startswith(('    ', '\t')):
            lines.append(blank_markup(line))
        else:
            lines.append(line)
    body = ''.join(lines)
    original, pieces, cursor = body, [], 0
    runs = list(re.finditer(r'`+', original))
    for opening in runs:
        if opening.start() < cursor or escaped(original, opening.start()):
            continue
        closing = next((candidate for candidate in runs if candidate.start() >= opening.end()
                        and len(candidate.group()) == len(opening.group())), None)
        if closing:
            pieces.extend((original[cursor:opening.start()], blank_markup(original[opening.start():closing.end()])))
            cursor = closing.end()
    pieces.append(original[cursor:])
    body = ''.join(pieces)
    # A backslash-escaped tag opener is displayed text, not an HTML anchor.
    # Encode only that opener; decode destinations only AFTER a genuine link
    # has been identified. Whole-document unescape would invent hidden anchors.
    return ''.join('&lt;' if char == '<' and escaped(body, i) else char for i, char in enumerate(body))


def rendered_links(body):
    """Recognize supported actual link markup, then decode its destination.

    HTML extraction is separate from Markdown text extraction. Markdown syntax
    in raw HTML/attributes is never mistaken for an active Markdown link. Only
    closed, visible, standalone <a href=...> anchors with href/title attributes
    qualify; hidden/nested/script/style/pre/code contexts cannot provide proof.
    """
    hrefs = []
    visible = visible_link_source(body)

    class Anchors(HTMLParser):
        def __init__(self):
            super().__init__(convert_charrefs=False)
            self.markup = False
            self.stack = []
            self.anchor = None

        def handle_starttag(self, tag, attrs):
            self.markup = True
            tag = tag.lower()
            if tag == 'a' and not self.stack:
                destinations = [value for key, value in attrs if key == 'href' and value is not None]
                allowed = len(destinations) == 1 and all(key in ('href', 'title') for key, _ in attrs)
                self.anchor = {'href':destinations[0], 'visible':False, 'paragraph':self.paragraph} if allowed else None
            elif self.anchor:
                # Restrict labels to ordinary text or unstyled visible inline
                # tags. An unsupported descendant invalidates the whole anchor.
                if tag not in ('em', 'strong', 'code') or attrs:
                    self.anchor = None
            if tag not in ('area', 'base', 'br', 'col', 'embed', 'hr', 'img', 'input', 'link', 'meta', 'param', 'source', 'track', 'wbr'):
                self.stack.append(tag)

        def handle_startendtag(self, tag, attrs):
            self.markup = True
            if self.anchor:
                self.anchor = None

        def handle_endtag(self, tag):
            self.markup = True
            tag = tag.lower()
            if self.stack and self.stack[-1] == tag:
                if tag == 'a' and len(self.stack) == 1:
                    if self.anchor and self.anchor['visible'] and self.anchor['paragraph'] == self.paragraph:
                        hrefs.append((self.anchor['href'], self.paragraph))
                    self.anchor = None
                self.stack.pop()
            else:
                self.anchor = None

        def handle_data(self, data):
            if self.anchor and data.strip():
                self.anchor['visible'] = True

        def handle_entityref(self, name):
            if self.anchor and html.unescape('&'+name+';').strip():
                self.anchor['visible'] = True

        def handle_charref(self, name):
            if self.anchor and html.unescape('&#'+name+';').strip():
                self.anchor['visible'] = True

        def handle_decl(self, decl):
            self.markup = True

        def handle_pi(self, data):
            self.markup = True

        def unknown_decl(self, data):
            self.markup = True

        def handle_comment(self, data):
            self.markup = True

    pattern = r'''(?<!!)\[[^\]\n]*\]\(\s*(?:<([^>\n]+)>|([^\s()\n]+))(?:\s+(?:"[^"]*"|'[^']*'|\([^)]*\)))?\s*\)'''
    parser = Anchors()
    for index, paragraph in enumerate(visible.split('\n\n')):
        # Keep HTML context across blank lines, including unclosed pre/code/
        # script containers. A partial raw tag is ambiguous, never Markdown.
        parser.markup = bool(parser.stack) or bool(re.search(r'<(?:/?[A-Za-z]|!)', paragraph))
        parser.paragraph = index
        parser.feed(paragraph + '\n\n')
        if not parser.markup:
            for match in re.finditer(pattern, paragraph):
                if not escaped(paragraph, match.start()):
                    hrefs.append((match.group(1) or match.group(2), index))
    parser.close()
    return [(html.unescape(href), index) for href, index in hrefs]


def rendered_hrefs(body):
    return [href for href, _ in rendered_links(body)]


def has_target_href(root, document, body, target, paragraph_index=None, section_number=None):
    """Resolve rendered links from document parent, retaining the repo boundary."""
    root = root.resolve()
    expected = (root / target).resolve()
    if root not in expected.parents or not expected.is_file():
        return False
    for href, index in rendered_links(body):
        if paragraph_index is not None and paragraph_index != index:
            continue
        if section_number is not None:
            headings = list(re.finditer(r'(?m)^## (\d+)\. [^\n]+\n', body))
            matches = [(i, head) for i, head in enumerate(headings) if int(head.group(1)) == section_number]
            if len(matches) != 1:
                continue
            i, head = matches[0]
            section_end = headings[i+1].start() if i+1 < len(headings) else len(body)
            paragraphs = body.split('\n\n')
            start = sum(len(p)+2 for p in paragraphs[:index])
            if not (head.end() <= start and start+len(paragraphs[index]) <= section_end):
                continue
        parts = urlsplit(href)
        if parts.scheme or parts.netloc or not parts.path or parts.path.startswith('/'):
            continue
        resolved = (root / Path(document).parent / unquote(parts.path)).resolve()
        if root in resolved.parents and resolved == expected and resolved.is_file():
            return True
    return False


def header_delta(root, path, version, enforcement=None):
    new, _ = doc(root, path)
    old, _ = doc(BASE, path)
    require(new['version'] == version, 'VERSION:' + path)
    logs = new.get('change_log', [])
    require(len(logs) == len(old['change_log']) + 1 and logs[1:] == old['change_log']
            and logs[0]['version'] == version, 'CHANGE_LOG:' + path)
    normalized = copy.deepcopy(new)
    normalized['version'] = old['version']
    normalized['change_log'] = old['change_log']
    if enforcement:
        require(new.get('enforcement') == enforcement, 'ENFORCEMENT:' + path)
        normalized['enforcement'] = old['enforcement']
    require(normalized == old, 'FRONTMATTER_SCOPE:' + path)

def lifecycle_data(features, phase, state, completion=False):
    oldfeatures = js(BASE, '.agent/feature_list_015.json')
    oldphase = yaml.safe_load(read(BASE, 'docs/planning/phase_015.yaml'))
    oldstate = js(BASE, '.agent/run_state.json')
    require(phase.get('status') == 'in_progress', 'PHASE_CONTINUITY')
    require([x['id'] for x in features['features']] == [x['id'] for x in oldfeatures['features']], 'FEATURE_IDS')
    require([x['id'] for x in phase['steps']] == [x['id'] for x in oldphase['steps']], 'STEP_IDS')
    for key in features:
        if key != 'features':
            require(features[key] == oldfeatures[key], 'FEATURE_ENVELOPE')
    for key in phase:
        if key != 'steps':
            require(phase[key] == oldphase[key], 'PHASE_ENVELOPE')
    for row, old in zip(features['features'], oldfeatures['features']):
        if row['id'] != '113':
            require(row == old, 'PREDECESSOR_SUCCESSOR:' + row['id'])
            continue
        allowed = {'status', 'passes', 'contract_ref', 'completed_at', 'qa', 'evidence'}
        require({k:v for k,v in row.items() if k not in allowed} == {k:v for k,v in old.items() if k not in allowed}, 'FEATURE113_IMMUTABLE')
        require(row['status'] in ('pending', 'in_progress', 'complete'), 'FEATURE113_STATUS')
        if row['status'] == 'complete':
            require(row.get('passes') is True and row.get('qa') == '.agent/qa/113.json'
                    and row.get('evidence') == '.agent/evidence/113/evidence-gate.txt'
                    and row.get('contract_ref') == '.agent/contracts/113.json' and row.get('completed_at'), 'COMPLETION_REFERENCES')
    feature = next(x for x in features['features'] if x['id'] == '113')
    for row, old in zip(phase['steps'], oldphase['steps']):
        if row['id'] != '113':
            require(row == old, 'STEP_HISTORY:' + row['id'])
        else:
            require(row.get('description') == old['description'], 'STEP113_DESCRIPTION')
            require(set(row) <= {'id', 'description', 'status', 'evidence', 'docs_updated', 'changelog_entry'}, 'STEP113_FIELDS')
            require(row['status'] == {'pending':'PENDING', 'in_progress':'IN_PROGRESS', 'complete':'COMPLETED'}[feature['status']], 'LIFECYCLE_AGREEMENT')
            if row['status'] == 'COMPLETED':
                require(row.get('evidence') == '.agent/evidence/113/evidence-gate.txt'
                        and row.get('docs_updated') and row.get('changelog_entry'), 'STEP113_COMPLETION')
    require(state['current_phase'] == oldstate['current_phase'] and state['current_feature'] in ('113', '114'), 'RUN_POSITION')
    budget = state['scope_budget']
    require(budget['original_estimate_lines'] == 7200 and budget['ceiling_130pct'] == 9360
            and budget['per_feature'][:3] == oldstate['scope_budget']['per_feature'][:3]
            and budget['total_lines_changed'] <= 9360, 'BUDGET_CONTINUITY')
    for key, old in oldstate['circuit_breaker'].items():
        now = state['circuit_breaker'].get(key)
        if not key.startswith('113-'):
            require(now == old, 'HISTORICAL_COUNTER:' + key)
        else:
            require(now and now['limit'] == old['limit'] == 3 and old['attempts'] <= now['attempts'] <= 3, 'COUNTER_CONTINUITY:' + key)
    require(state['history'][:len(oldstate['history'])] == oldstate['history'], 'HISTORY_APPEND_ONLY')
    require(completion is False or feature['status'] == 'complete', 'COMPLETION_PROJECTION')

#!/usr/bin/env python3
"""Disclosed Parent adapter: authenticate feature113 evidence before closure."""
import hashlib
import json
from pathlib import Path
import re
import stat
import subprocess

import jsonschema

ROOT = Path(__file__).resolve().parents[3]
E = '.agent/evidence/113/'


def read(path):
    p = ROOT / path
    assert not p.is_symlink() and p.is_file(), path
    return p.read_bytes()


def obj(path):
    return json.loads(read(path))


def digest(data):
    return hashlib.sha256(data).hexdigest()


def approved(record):
    assert record['final_verdict']['approved'] is True
    for key in ('issues', 'missing_acceptance_coverage', 'unsupported_claims'):
        assert record[key] == [], key


def git(*args):
    return subprocess.check_output(['git', '-C', str(ROOT), *args])


def receipt(path, command, candidate):
    raw = read(path).decode()
    assert raw.startswith('COMMAND: ' + command + '\n'), path
    assert 'CANDIDATE_SHA: ' + candidate + '\n' in raw, path
    assert re.findall(r'^EXIT:\s*(-?\d+)\s*$', raw, re.M) == ['0'], path
    assert raw.rstrip().endswith('EXIT: 0'), path


def replay(files, candidate):
    for path, expected in files.items():
        assert not Path(path).is_absolute() and '..' not in Path(path).parts, path
        assert re.fullmatch('[0-9a-f]{64}', expected['sha256']), path
        assert expected['mode'] in ('100644', '100755'), path
        assert digest(read(path)) == expected['sha256'], path
        mode = '100755' if stat.S_IMODE((ROOT / path).stat().st_mode) & 0o111 else '100644'
        assert mode == expected['mode'], path + ' current mode'
        assert digest(git('show', candidate + ':' + path)) == expected['sha256'], path + ' Git replay'
        entry = git('ls-tree', candidate, '--', path).decode().split('\t')[0].split()
        assert len(entry) == 3 and entry[:2] == [expected['mode'], 'blob'], path + ' Git mode'


def main():
    contract = obj('.agent/contracts/113.json')
    qa = obj('.agent/qa/113.json')
    fixed = obj(E + 'fixed-inputs.json')
    source = obj(E + 'current-source-inputs.json')
    source_seal = obj(E + 'current-source-seal.json')
    mode_b = obj('.agent/validator/113-mode-b.json')
    semantic = obj(E + 'source-review.json')
    bundle = obj('.agent/validator/113-test-bundle-review.json')
    for record in (qa, mode_b, semantic, bundle):
        approved(record)
    jsonschema.validate(qa, obj('.nizam/schema/qa_verdict.schema.json'))
    jsonschema.validate(contract, obj('.nizam/schema/contract.schema.json'))
    assert contract['status'] == 'approved'
    assert qa['feature_id'] == '113' and qa['qa_pass'] is True and qa['required_fixes'] == []
    assert len(contract['verification']) == 16
    assert qa['checks_run'] == qa['checks_passed'] == len(qa['checks']) == 16
    fixed_hash = digest(read(E + 'fixed-inputs.json'))
    source_hash = digest(read(E + 'current-source-inputs.json'))
    assert obj(E + 'fixed-inputs-seal.json')['manifest_sha256'] == fixed_hash
    assert source_seal['manifest_sha256'] == source_hash
    assert source['fixed_inputs_sha256'] == fixed_hash
    contract_hash = digest(read('.agent/contracts/113.json'))
    assert source['contract_sha256'] == fixed['contract_sha256'] == contract_hash
    candidate = source['candidate']['commit']
    assert re.fullmatch('[0-9a-f]{40}', candidate)
    assert source_seal['candidate_commit'] == candidate
    assert git('rev-parse', candidate + '^{tree}').decode().strip() == source['candidate']['tree']
    assert subprocess.run(['git', '-C', str(ROOT), 'merge-base', '--is-ancestor', candidate, 'HEAD'], capture_output=True).returncode == 0
    actors = fixed['actors']
    actor_values = [actors[k] for k in ('generator', 'test_author', 'bundle_reviewer', 'final_qa')]
    assert all(isinstance(actor, str) and actor for actor in actor_values) and len(set(actor_values)) == 4
    assert qa['actor'] == semantic['actor'] == actors['final_qa']
    assert mode_b['actor'] == bundle['actor'] == actors['bundle_reviewer']
    bundle_path = '.agent/validator/113-test-bundle-review.json'
    bundle_hash = digest(read(bundle_path))
    assert qa['bundle_review_sha256'] == bundle_hash
    assert digest(git('show', candidate + ':' + bundle_path)) == bundle_hash
    for record in (qa, semantic, mode_b):
        assert record['candidate_commit'] == candidate
        assert record['source_manifest_sha256'] == source_hash
        assert record['fixed_inputs_sha256'] == fixed_hash
    assert qa['mode_b_sha256'] == digest(read('.agent/validator/113-mode-b.json'))
    assert len(source['source_paths']) == 8
    replay(fixed['files'], candidate)
    replay(source['source_paths'], candidate)
    expected_evidence = [E + 'qa-%02d.txt' % i for i in range(1, 17)]
    for required, actual, path in zip(contract['verification'], qa['checks'], expected_evidence):
        assert actual['command'] == required['command'] and actual['evidence'] == path
        assert actual['exit_code'] == 0
        receipt(path, required['command'], candidate)
    spot = E + '113-qa-adversarial.txt'
    assert qa['adversarial']['performed'] is True and qa['adversarial']['evidence'] == spot
    assert qa['adversarial']['command'] not in [x['command'] for x in contract['verification']]
    receipt(spot, qa['adversarial']['command'], candidate)
    assert len(qa['evidence_files']) == 17 and set(qa['evidence_files']) == set(expected_evidence + [spot])
    assert semantic['evidence'] and all(read(path) for path in semantic['evidence'])
    print('EVIDENCE GATE PASS: feature113 actual sealed Git candidate; sixteen independent checks, semantic review, own adversarial receipt; fixed/source bytes and modes replay exactly.')

# Universal Circuit Breaker (DD-2)

## 1. Overview

This document is the single source of truth for the framework's mandatory
3-strike / three-attempt limit — Design Decision DD-2, remediating the
"infinite loop" failure mode where an agent retries a failing step indefinitely
without ever escalating. Every execution loop that can repeat a step on failure
— contract revision, implementation rework, or any other repair cycle defined
elsewhere in this framework — embeds this exact limit. No document elsewhere in
the framework restates or redefines these mechanics; they cross-reference this
one by path.

## 2. Scope: Every Repeatable Loop, No Exceptions

This breaker applies uniformly to every step in the framework capable of being
retried after a failure, including but not limited to:

- Loop 1 contract revisions (`01_execution.md` Section 2) — a Generator revising
  a proposed contract after Validator Mode A or Evaluator rejection.
- Loop 2 implementation reworks (`01_execution.md` Section 3) — a Generator
  reworking code after a Validator Mode B rejection, or after a failing QA
  verdict.
- Any subagent retry a pipeline performs after a malformed or empty subagent
  response.

A step not explicitly named above is not exempt by omission — "any single,
repeatable step" is the governing scope, not an enumerated allowlist.

## 3. The Per-Attempt Strategy Table

Each attempt at a failing step MUST follow an escalating strategy. An agent
MUST NOT repeat the same diagnostic approach across attempts — each attempt
number below prescribes a materially different investigative posture:

| Attempt | Strategy |
|---|---|
| **1** | **Direct fix.** Read the exact error output. Fix the specific line or command it identifies. Assume the defect is local and mechanical. |
| **2** | **Type/interface analysis.** If the direct fix did not resolve the failure, step back and check for a structural mismatch — a type signature, an interface contract, a schema shape, or an assumption about another component's output that does not actually hold. |
| **3** | **Architectural review.** If the failure persists after a structural fix attempt, question the approach itself. Consider whether the chosen implementation strategy, not merely its details, is what needs to change. |
| **4+** | **FORBIDDEN.** No fourth attempt at the same step is permitted, regardless of how promising a new idea seems. See Section 4. |

Skipping a strategy tier (for example, attempting an architectural rewrite on
attempt 1 without first trying the direct fix) is itself a protocol violation —
the escalating order exists so that cheap, likely fixes are exhausted before
expensive, uncertain ones are attempted.

### 3.1 Attempt isolation

Every attempt MUST start from the last approved immutable base in its own
attempt-scoped workspace. In Git, use a detached worktree at a newly-created,
validated temporary path (conceptually `git worktree add --detach
<attempt-worktree> <approved-commit>`); a non-Git system uses an equivalent
copy-on-write snapshot. Never run a retry in the shared developer worktree.

The attempt root and approved base are recorded before edits begin. A failed
attempt's evidence is captured from that root and then only that exact root is
removed through the worktree/snapshot manager. A successful attempt's reviewed
diff or commit is the only material promoted back to the feature branch. The
shared tree is never recovered with repository-wide reset or clean commands.

## 4. Attempt 4 Is Forbidden — The Breach Procedure

If a step's third attempt (per Section 3's table) also fails, the circuit
breaker has tripped. The acting agent MUST, in order:

1. **Capture and quarantine the third attempt.** Write its diagnostics, diff,
   and exact attempt-root/base identifiers under `.agent/evidence/`. Then ask
   the worktree or snapshot manager to remove **only that validated attempt
   root**. No shared-worktree reset, repository clean, glob, or broad path is
   permitted. If the root cannot be proved attempt-owned, leave it in place,
   record it, and escalate rather than guessing at cleanup.
2. **Set the canonical step status to `BLOCKED`, then repair derived state.**
   The phase document's step-level `status` field, per
   `schema/phase.schema.json`, is the lifecycle source of truth and is written
   first. `.agent/run_state.json` is coordination state derived from that
   canonical value and is updated in a separate, idempotent operation. If the
   files disagree, execution halts; rebuild the derived entry from the phase
   document and validate both artifacts before continuing. The protocol never
   claims two files share one atomic filesystem write.
3. **Log the failure** to the technical-debt register (conventionally
   `docs/planning/DEBT.md`), naming at minimum: the phase, the feature, the
   failure type, and a reference to the last attempt's response or evidence
   file.
4. **Escalate to a human and terminate.** The agent stops. Autonomous
   continuation past a third failed attempt on any single step is never
   permitted, under any framing ("just one more try," "I think I see it now")
   — that framing is precisely what this document exists to override.

Attempt 4 is not a slower, more cautious version of attempts 1-3. It is not
permitted to occur at all.

## 5. Where Breaker State Lives

Every in-flight attempt counter is tracked in `.agent/run_state.json`'s
`circuit_breaker` object (`schema/run_state.schema.json`), keyed by a
step-identifying string (for example `"001-implementation"` or
`"002-contract"`), each holding:

```json
{
  "circuit_breaker": {
    "<step-key>": {
      "attempts": 2,
      "limit": 3
    }
  }
}
```

An agent beginning any retryable step MUST first reconcile this derived object
with the canonical phase step, then read the counter to determine which attempt
it is about to make. It MUST increment the relevant counter idempotently after
each attempt concludes (success or failure). If reconciliation cannot prove the
derived entry corresponds to the canonical step, the pipeline halts and rebuilds
it before another attempt. An agent that begins a fourth attempt without having
respected this counter has violated Section 4 regardless of its outcome.

## 6. Escalation Protocol

Escalation (Section 4, step 4) means the pipeline halts and a human reviewer is
notified — it does not mean the pipeline silently proceeds to the next feature
while marking the blocked one aside for "later." A `BLOCKED` feature remains
blocked until a human either:

- Provides a corrected approach or additional context that resolves the
  underlying defect, after which the step may be attempted fresh (resetting the
  attempt counter to 0), or
- Explicitly authorizes cancelling or descoping the feature, after which its
  status moves to `cancelled` (which, per the Dependency Enforcement Rule in
  `00_planning.md` Section 5, satisfies any downstream feature that depended on
  it, exactly as `complete` would).

No agent may unilaterally decide a `BLOCKED` feature should be silently skipped
or reattempted without one of the two human actions above.

## 7. References

- `01_execution.md` — names every contract-revision and implementation-rework
  loop this breaker bounds.
- `00_planning.md` Section 5 — the Dependency Enforcement Rule that governs how
  a `BLOCKED` or `cancelled` feature affects downstream feature eligibility.
- `schema/run_state.schema.json` — the structural definition of the
  `circuit_breaker` object referenced in Section 5.
- `schema/phase.schema.json` — the step-level `status` enum (including
  `BLOCKED`) a phase document's steps carry.
